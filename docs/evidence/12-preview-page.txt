import { getAdminPostById } from '../../../../../../server/services/blogService';
import { notFound, redirect } from 'next/navigation';
import { auth } from '../../../../../../auth';

export const metadata = {
  robots: { index: false, follow: false }
};

export default async function PreviewPage(props: { params: Promise<{ id: string }> }) {
  const params = await props.params;
  const session = await auth();
  if (!session) redirect('/admin/login');

  if (!params.id) notFound();
  const post = await getAdminPostById(params.id);
  if (!post) notFound();

  return (
    <article className="container mx-auto px-4 py-16 max-w-3xl bg-yellow-50">
      <div className="mb-4 text-orange-600 font-bold">PREVIEW MODE</div>
      <h1 className="text-5xl font-bold mb-6">{post.title}</h1>
      <div className="prose prose-lg max-w-none whitespace-pre-wrap">
        {post.body}
      </div>
    </article>
  );
}
